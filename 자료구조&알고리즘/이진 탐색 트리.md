BST(binary search tree)

이진 트리는 각 노드에 자식이 0개, 1개 또는 2개다.
이진 탐색 트리는 여기에 아래 규칙이 추가되었다.
- 각 노드의 자식은 최대 왼쪽에 하나, 오른쪽에 하나
- 노드의 왼쪽 자손은 노드보다  작은 값만 포함 가능
- 노드의 오른쪽 자손은 노드보다 큰 값만 포함 가능

| 연산          |   최선 |       평균 |   최악 |
| ----------- | ---: | -------: | ---: |
| 읽기(특정 값 접근) | O(1) | O(log n) | O(n) |
| 검색          | O(1) | O(log n) | O(n) |
| 삽입          | O(1) | O(log n) | O(n) |
| 삭제          | O(1) | O(log n) | O(n) |
| 순회          | O(n) |     O(n) | O(n) |
```java
public class BinarySearchTree{
	// 노드 클래스
	private static class Node{
		int data;
		Node left;
		Node right;
		
		Node(int data){
			this.data = data;
			this.left = null;
			this.right = null;
		}
	}
	// 트리의 노드
	private Node root;
	
	// 삽입
	public void insert(int data){
		root = insert(root, data);
	}
	private Node insert(Node node, int data){
		// 빈 위치를 찾으면 새 노드 생성
		if(node == null){
			return new Node(data);
		}
		// 현재 노드보다 작으면 왼쪽
		if(data < node.data){
			node.left = insert(node.left, data);
		}else if{
		// 현재 노드보다 크면 오른쪽
			node.right = insert(node.right, data);
		}
		
		// 중복값은 삽입하지 않음
		return node;
	}
	
	// 검색
	public boolean contains(int data){
		return contains(root, data);
	}
	private boolean contains(Node node, int data){
		// 찾지 못한 경우
		if(node == null){
			return false;
		}
		// 찾은 경우
		if(data == node.data){
			return true;
		}
		// 작으면 왼쪽 탐색
		if(data < node.data){
			return contains(node.left,data);
		}
		// 크면 오른쪽 탐색
		if(data > node.data){
			return contains(node.right, data);
		}
	}
	
	// 삭제
	public void remove(int data){
		root = remove(root, data);
	}
	private Node remove(Node node, int data){
		if(node == null){
			return null;
		}
		// 삭제할 값이 더 적은 겨우
		if(data < node.data){
			node.left = remove(node.left, data);
		}else if(data > node.data){
		// 삭제할 값이 더 큰 경우
			node.right = remove(node.right, data);
		}else{
		// 삭제할 노드를 찾은 경우
			// 자식이 없는 경우 또는 오른쪽 자식만 있는 경우
			if(node.left == null){
				return node.right;
			}
			// 왼쪽 자식만 있는 경우
			if(node.right == null){
				return node.left;
			}
			// 자식이 2개인 경우
			// 오른쪽 서브트리에서 가장 적은 값 찾기
			Node successor = findMin(node.right);
			// 현재 노드의 값을 후계자 값으로 변경
			node.data = successor.data;
			// 후계자 노드 삭제
			node.right = remove(node.right, successor.data)
		}
		return node;
	}
	// 가장 작은 노드 찾기
	private Node findMin(Node node){
		while(node.left != null){
			node = node.left;
		}
		return node;
	}
	// 중위 순회
	public void inorder(){
		inorder(root);
		System.out.println();
	}
	private void inorder(Node node){
		if(node == null){
			return;
		}
		inorder(node.left);
		System.out.println(node.data + " ");
		inorder(node.right);
	}
	
	public static void main(String[] args) { 
		BinarySearchTree tree = new BinarySearchTree(); 
		tree.insert(50); 
		tree.insert(30); 
		tree.insert(70); 
		tree.insert(20); 
		tree.insert(40); 
		tree.insert(60); 
		tree.insert(80); 
		System.out.print("중위 순회: "); 
		tree.inorder(); 
		System.out.println("40 존재 여부: " + tree.contains(40)); 
		System.out.println("100 존재 여부: " + tree.contains(100)); 
		tree.remove(70); 
		System.out.print("70 삭제 후: "); 
		tree.inorder(); 
	}
}
```

이진 탐색 트리에 데이터를 삽입할 때 정렬된 데이터로 트리를 생성하면 불균형이 심해질 수 있다.
무작위로 정렬된 데이터로 트리를 생성해야 대개 균형 잡힌 트리가 생성된다.
