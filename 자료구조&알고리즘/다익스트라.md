다익스트라(Dijkstra)는 **가중치가 있는 그래프에서 한 시작점으로부터 다른 모든 정점까지의 최단거리**를 구하는 알고리즘

>note. 음의 가중치를 허용하지 않음

[[배열]]을 이용한 구현 코드
```java
import java.util.*;

public class Solution {

    // 도시
    static class City {

        String name;

        // 인접 도시와 이동 비용
        Map<City, Integer> routes = new HashMap<>();

        City(String name) {
            this.name = name;
        }

        // 현재 도시 -> 다른 도시로 가는 경로 추가
        void addRoute(City city, int price) {
            routes.put(city, price);
        }
    }


    static List<String> dijkstraShortestPath(
            City startingCity,
            City finalDestination
    ) {

        // 시작 도시에서 각 도시까지의 현재 최저 비용
        Map<String, Integer> cheapestPricesTable = new HashMap<>();

        // 최단 경로에서 각 도시의 이전 도시
        Map<String, String> cheapestPreviousStopoverCityTable
                = new HashMap<>();

        // 아직 방문하지 않은 도시
        List<City> unvisitedCities = new ArrayList<>();

        // 방문한 도시
        Set<String> visitedCities = new HashSet<>();


        // 시작 도시까지의 비용은 0
        cheapestPricesTable.put(startingCity.name, 0);

        City currentCity = startingCity;


        // 방문할 도시가 있는 동안 반복
        while (currentCity != null) {

            // 현재 도시 방문 처리
            visitedCities.add(currentCity.name);

            // 미방문 목록에서 현재 도시 제거
            unvisitedCities.remove(currentCity);


            // 현재 도시와 연결된 모든 인접 도시 확인
            for (Map.Entry<City, Integer> entry
                    : currentCity.routes.entrySet()) {

                City adjacentCity = entry.getKey();

                int price = entry.getValue();


                // 아직 방문하지 않은 도시라면
                // unvisitedCities에 추가
                if (!visitedCities.contains(adjacentCity.name)
                        && !unvisitedCities.contains(adjacentCity)) {

                    unvisitedCities.add(adjacentCity);
                }


                // 시작 도시 -> 현재 도시 -> 인접 도시 비용
                int priceThroughCurrentCity =
                        cheapestPricesTable.get(currentCity.name)
                                + price;


                // 아직 해당 도시까지의 비용을 모르는 경우
                // 또는 기존 비용보다 더 저렴한 경로를 찾은 경우
                if (!cheapestPricesTable.containsKey(adjacentCity.name)
                        || priceThroughCurrentCity
                        < cheapestPricesTable.get(adjacentCity.name)) {

                    // 최저 비용 갱신
                    cheapestPricesTable.put(
                            adjacentCity.name,
                            priceThroughCurrentCity
                    );

                    // 이 도시로 오기 직전의 도시 기록
                    cheapestPreviousStopoverCityTable.put(
                            adjacentCity.name,
                            currentCity.name
                    );
                }
            }


            // 아직 방문하지 않은 도시 중
            // 시작점에서 가장 저렴하게 갈 수 있는 도시 선택
            currentCity = null;

            int cheapestPrice = Integer.MAX_VALUE;

            for (City city : unvisitedCities) {

                int price = cheapestPricesTable.get(city.name);

                if (price < cheapestPrice) {

                    cheapestPrice = price;

                    currentCity = city;
                }
            }
        }


        // =========================
        // 최단 경로 복원
        // =========================

        List<String> shortestPath = new ArrayList<>();

        String currentCityName = finalDestination.name;


        // 목적지에서 시작점까지 역추적
        while (!currentCityName.equals(startingCity.name)) {

            shortestPath.add(currentCityName);

            currentCityName =
                    cheapestPreviousStopoverCityTable.get(currentCityName);
        }


        // 시작 도시 추가
        shortestPath.add(startingCity.name);

        // 목적지 -> 시작점 순서였으므로 뒤집기
        Collections.reverse(shortestPath);

        return shortestPath;
    }


    public static void main(String[] args) {

        City atlanta = new City("Atlanta");
        City boston = new City("Boston");
        City chicago = new City("Chicago");
        City denver = new City("Denver");
        City elPaso = new City("El Paso");


        // 교재의 그래프 구성
        atlanta.addRoute(boston, 100);
        atlanta.addRoute(denver, 160);

        boston.addRoute(chicago, 120);
        boston.addRoute(denver, 180);

        chicago.addRoute(elPaso, 80);

        denver.addRoute(chicago, 40);
        denver.addRoute(elPaso, 140);


        List<String> shortestPath =
                dijkstraShortestPath(atlanta, elPaso);

        System.out.println(shortestPath);
    }
}
```

[[우선순위 큐]]를 이용한 구현

```java
import java.util.*;

public class Solution {

    // 도시
    static class City {

        String name;

        // 인접 도시 -> 이동 비용
        Map<City, Integer> routes = new HashMap<>();

        City(String name) {
            this.name = name;
        }

        void addRoute(City city, int price) {
            routes.put(city, price);
        }
    }


    // PriorityQueue에 저장할 객체
    static class State {

        City city;

        // 시작 도시에서 현재 도시까지의 비용
        int cost;

        State(City city, int cost) {
            this.city = city;
            this.cost = cost;
        }
    }


    static List<String> dijkstraShortestPath(
            City startingCity,
            City finalDestination
    ) {

        // 시작점에서 각 도시까지 현재까지 알려진 최저 비용
        Map<String, Integer> cheapestPricesTable = new HashMap<>();

        // 최단 경로에서 현재 도시 바로 이전에 방문한 도시
        Map<String, String> cheapestPreviousStopoverCityTable
                = new HashMap<>();


        // 비용이 작은 State가 먼저 나오도록 설정
        PriorityQueue<State> pq =
                new PriorityQueue<>(
                        (a, b) -> Integer.compare(a.cost, b.cost)
                );


        // 시작 도시의 비용은 0
        cheapestPricesTable.put(startingCity.name, 0);

        pq.offer(new State(startingCity, 0));


        while (!pq.isEmpty()) {

            // 현재까지 비용이 가장 작은 도시를 꺼냄
            State currentState = pq.poll();

            City currentCity = currentState.city;
            int currentCost = currentState.cost;


            // 이미 더 짧은 경로가 발견된 오래된 정보라면 무시
            if (currentCost
                    > cheapestPricesTable.get(currentCity.name)) {
                continue;
            }


            // 목적지에 도착했다면 더 탐색할 필요 없음
            if (currentCity == finalDestination) {
                break;
            }

            // 현재 도시의 모든 인접 도시 확인
            for (Map.Entry<City, Integer> entry
                    : currentCity.routes.entrySet()) {

                City adjacentCity = entry.getKey();

                int price = entry.getValue();


                // 시작점 -> 현재 도시 -> 인접 도시의 비용
                int priceThroughCurrentCity =
                        currentCost + price;


                // 기존에 알려진 비용
                int oldPrice =
                        cheapestPricesTable.getOrDefault(
                                adjacentCity.name,
                                Integer.MAX_VALUE
                        );


                // 현재 도시를 거쳐 가는 것이 더 저렴하다면 갱신
                if (priceThroughCurrentCity < oldPrice) {

                    // 최저 비용 갱신
                    cheapestPricesTable.put(
                            adjacentCity.name,
                            priceThroughCurrentCity
                    );

                    // 이전 도시 기록
                    cheapestPreviousStopoverCityTable.put(
                            adjacentCity.name,
                            currentCity.name
                    );

                    // 갱신된 경로를 PriorityQueue에 추가
                    pq.offer(
                            new State(
                                    adjacentCity,
                                    priceThroughCurrentCity
                            )
                    );
                }
            }
        }


        // ==========================
        // 최단 경로 복원
        // ==========================

        List<String> shortestPath = new ArrayList<>();

        String currentCityName = finalDestination.name;


        // 목적지까지 갈 수 없는 경우
        if (!cheapestPricesTable.containsKey(finalDestination.name)) {
            return shortestPath;
        }


        // 목적지 -> 시작점 순서로 역추적
        while (!currentCityName.equals(startingCity.name)) {

            shortestPath.add(currentCityName);

            currentCityName =
                    cheapestPreviousStopoverCityTable.get(
                            currentCityName
                    );
        }


        // 시작 도시 추가
        shortestPath.add(startingCity.name);

        // 시작점 -> 목적지 순서로 뒤집기
        Collections.reverse(shortestPath);

        return shortestPath;
    }


    public static void main(String[] args) {

        City atlanta = new City("Atlanta");
        City boston = new City("Boston");
        City chicago = new City("Chicago");
        City denver = new City("Denver");
        City elPaso = new City("El Paso");


        // 교재의 그래프
        atlanta.addRoute(boston, 100);
        atlanta.addRoute(denver, 160);

        boston.addRoute(chicago, 120);
        boston.addRoute(denver, 180);

        chicago.addRoute(elPaso, 80);

        denver.addRoute(chicago, 40);
        denver.addRoute(elPaso, 140);


        List<String> shortestPath =
                dijkstraShortestPath(atlanta, elPaso);

        System.out.println(shortestPath);
    }
}
```

# 대표 문제 SWEA 1249

### 문제 설명

`N × N` 크기의 지도에서 `(0,0)`부터 `(N-1,N-1)`까지 상하좌우로 이동한다.
각 칸에는 해당 지역을 복구하는 데 필요한 시간이 `0 ~ 9`로 주어진다.
이동하면서 방문한 지역의 복구 시간을 누적했을 때 도착점까지 가기 위해 필요한 **최소 복구 시간**을 구하는 문제이다.

## 구현
```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.PriorityQueue;

public class Solution {
	static int N;
	static int[][] map;
	static int[][] dist;
	
	static int[] dr = {-1, 1, 0, 0};
	static int[] dc = {0, 0, -1, 1};
	
	static class Node implements Comparable<Node>{
		int r;
		int c;
		int cost;
		
		Node(int r, int c, int cost){
			this.r = r;
			this.c = c;
			this.cost = cost;
		}
		
		@Overrie
		public int compareTo(Node other){
			return Integer.compare(this.cost, other.cost);
		}
	}
	
	public static void main(String[] args) throws Exception{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();
		int T = Integer.parseInt(br.readLine());
		
		for(int tc = 1; tc<=T; tc++){
			N = Integer.parseInt(br.readLine());
			map = new int[N][N];
			dist = new int[N][N];
			
			for(int r = 0; r < N; r++){
				String line = br.readLine();
				
				for(int c = 0; c < N; c++){
					map[r][c] = line.charAt(c) - '0';
				}
			}
			
			for(int r= 0; r < N; r++){
				Arrays.fill(dist[r], Integer.MAX_VALUE);
			}
			
			int answer = dijkstra();
			
			sb.append("#").append(tc).append(" ").append(answer).append("\n");
		}
		System.out.print(sb);
	}
	
	static int dijkstra(){
		PriorityQueue<Node> pq = new PriorityQueue<>();
		dist[0][0] = 0;
		pq.offer(new Node(0,0,0));
		
		while(!pq.isEmpty()){
			Node current = pq.poll();
			
			// 이미 더 짧은 경로가 발견된 경우
			if(current.cost > dist[current.r][current.c]){
				continue;
			}
			// 목적지에 도착하면 최소 비용 확정
			if(current.r == N - 1 && current.c == N -1){
				return current.cost
			}
			
			for(int d = 0; d < 4; d++){
				int nr = current.r + dr[d];
				int nc = current.c + dc[d];
				
				if(nr < 0 || nr >= N || nc < 0 || nc >= N){
					continue;
				}
				
				// 다음 칸의 복구 시간을 비용으로 추가
				int newCost = current.cost + map[nr][nc];
				
				if(newCost < dist[nr][nc]){
					dist[nr][nc] = newCost;
					pq.offer(new Node(nr,nc, newCost));
				}
			}
		}
		return dist[N - 1][N - 1];
	}
}
```
