다익스트라(Dijkstra)는 **가중치가 있는 그래프에서 한 시작점으로부터 다른 모든 정점까지의 최단거리**를 구하는 알고리즘

>note. 음의 가중치를 허용하지 않음

## 구현 코드
인접리스트 + [[우선순위 큐]] 기반 

```java
import java.io.*;
import java.util.*;

public class Solution {
	static class Edge{
		int to;
		int cost;
		
		Edge(int to, int cost){
			this.to = to;
			this.cost = cost;
		}
	}
	
	static class Node implements Comparable<Node>{
		int vertex;
		int cost;
		
		Node(int vertex, int cost){
			this.vertex = vertex;
			this.cost = cost;
		}
		
		@Override
		public int compareTo(Node other){
			return Integer.compare(this.cost, this.other);
		}
	}
	
	static final int INF = Integer.MAX_VALUE;
	
	static int V;
	static int E;
	
	static List<Edge>[] graph;
	// 시작점에서 각 정점까지의 최소 비용
	static int[] dist;
	// 최단 경로에서 해당 정점 바로 직전의 정점
	static int[] prev;
	
	static void dijkstra(int start){
		PriorityQueue<Node> pq = new PriorityQueue<>();
		Arrays.fill(dist, INF);
		Arrays.fill(prev, -1);
		
		// 시작 정점
		dist[start] = 0;
		pq.offer(new Node(start, 0));
		
		while(!pq.isEmpty()){
			// 현재까지 시작점에서 가장 가까운 정점
			Node current = pq.poll();
			
			int now = current.vertex;
			int currentCost = current.cost;
			
			// 이미 더 짧은 경로가 발견된 경우
			if(currentCost > dist[now]){
				continue;
			}
			
			// 현재 정점과 연결된 모든 간선 확인
			for(Edge edge : graph[now]){
				int next = edge.to;
				
				// 현재 정점을 거쳐 next로 가는 비용
				int newCost = dist[now] + edge.cost;
				
				// 기존 경로보다 더 짧다면 갱신
				if(newCost < dist[next]){
					dist[next] = newCost;
					prev[next] = now;
					// note. 이때 같은 정점 Node가 중복되어 pq에 포함될 수 있음
					// 어차피 가장 최소 값을 가지는 Node 제외하고 skip 됨
					pq.offer(new Node(next, newCost));
				}
			}	
		}
	}  
	
	// start -> destination 최단 경로 복원
	static List<Integer> getPath(int start, int destination){
		List<Integer> path = new ArrayList<>();
		
		// 도달할 수 없는 경우
		if(dist[destination] == INF){
			return path;
		}
		
		int current = destination;
		
		// 목적지부터 이전 정점을 따라 역추적
		while(current != -1){
			path.add(current);
			
			if(current == start){
				break;
			}
			
			current = prev[current];
		}
		
		// 목적지 -> 시작점 순서이므로 뒤집기
		Collections.reverse(path);
		return path;
	}
	
	public static void main(String[] args) throws Exception{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		
		V = Integer.parseInt(st.nextToken());
		E = Integer.parseInt(st.nextToken());
		
		graph = new ArrayList[V + 1];
		dist = new int[V + 1];
		
		for(int i = 1; i <= V; i++){
			graph[i] = new ArrayList<>();
		}
		/*
		 * 입력 형식 예시
		 * V E
		 * from to cost
		 * from to cost
		 */
			 
		 for(int i = 0; i < E; i++){
			st = new StringTokenizer(br.readLine());
			
			int from = Integer.parseInt(st.nextToken());
			int to = Integer.parseInt(st.nextToken());
			int cost = Integer.parseInt(st.nextToken());
			
			// 방향 그래프
			graph[from].add(new Edge(to, cost));
			
			// 무향 그래프 일시 아래 라인 추가
			// graph[to].add(new Edge(from, cost));	
		}
		
		int start = 1;
		int destination = V;
		
		dijkstra(start);
		
		// 최소 비용
		if(dist[destination] == INF){
			System.out.println("도달 할 수 없음");
		}else{
			System.out.println("최소 비용: " + dist[destination]);
			
			// 실제 경로
			List<Integer> path = getPath(start, destination);
			System.out.print("최단 경로: ");
			
			for(int i = 0; i < path.size(); i++){
				System.out.print(path.get(i));
				if(i < path.size() - 1){
					System.out.print(" -> ");
				}
			}
			System.out.println();
		}
	}
}
```


# 대표 문제 SWEA 1249

### 문제 설명

`N × N` 크기의 지도에서 `(0,0)`부터 `(N-1,N-1)`까지 상하좌우로 이동한다.
각 칸에는 해당 지역을 복구하는 데 필요한 시간이 `0 ~ 9`로 주어진다.
이동하면서 방문한 지역의 복구 시간을 누적했을 때 도착점까지 가기 위해 필요한 **최소 복구 시간**을 구하는 문제이다.

## 구현
```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.PriorityQueue;

public class Solution {
	static int N;
	static int[][] map;
	static int[][] dist;
	
	static int[] dr = {-1, 1, 0, 0};
	static int[] dc = {0, 0, -1, 1};
	
	static class Node implements Comparable<Node>{
		int r;
		int c;
		int cost;
		
		Node(int r, int c, int cost){
			this.r = r;
			this.c = c;
			this.cost = cost;
		}
		
		@Overrie
		public int compareTo(Node other){
			return Integer.compare(this.cost, other.cost);
		}
	}
	
	public static void main(String[] args) throws Exception{
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();
		int T = Integer.parseInt(br.readLine());
		
		for(int tc = 1; tc<=T; tc++){
			N = Integer.parseInt(br.readLine());
			map = new int[N][N];
			dist = new int[N][N];
			
			for(int r = 0; r < N; r++){
				String line = br.readLine();
				
				for(int c = 0; c < N; c++){
					map[r][c] = line.charAt(c) - '0';
				}
			}
			
			for(int r= 0; r < N; r++){
				Arrays.fill(dist[r], Integer.MAX_VALUE);
			}
			
			int answer = dijkstra();
			
			sb.append("#").append(tc).append(" ").append(answer).append("\n");
		}
		System.out.print(sb);
	}
	
	static int dijkstra(){
		PriorityQueue<Node> pq = new PriorityQueue<>();
		dist[0][0] = 0;
		pq.offer(new Node(0,0,0));
		
		while(!pq.isEmpty()){
			Node current = pq.poll();
			
			// 이미 더 짧은 경로가 발견된 경우
			if(current.cost > dist[current.r][current.c]){
				continue;
			}
			// 목적지에 도착하면 최소 비용 확정
			if(current.r == N - 1 && current.c == N -1){
				return current.cost
			}
			
			for(int d = 0; d < 4; d++){
				int nr = current.r + dr[d];
				int nc = current.c + dc[d];
				
				if(nr < 0 || nr >= N || nc < 0 || nc >= N){
					continue;
				}
				
				// 다음 칸의 복구 시간을 비용으로 추가
				int newCost = current.cost + map[nr][nc];
				
				if(newCost < dist[nr][nc]){
					dist[nr][nc] = newCost;
					pq.offer(new Node(nr,nc, newCost));
				}
			}
		}
		return dist[N - 1][N - 1];
	}
}
```
