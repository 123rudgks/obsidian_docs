
스택이 데이터를 저장하는 방법은 [[배열]]과 같다. 즉, 단순히 원소들의 리스트다. 다만 한 가지, 스택에는 다음과 같은 세 가지 제약이 있다.
- 데이터는 스택의 끝에만 삽입 가능
- 데이터는 스택의 끝에서만 삭제 가능
- 스택의 마지막 원소만 읽을 수 있음

>[!note]
>배열은 대부분 프로그래밍 언어에 내장되어있고 컴퓨터 메모리와 직접 상호작용을 하지만 스택은 **추상 데이터 타입**이므로 배열을 쓰든 다른 내장 데이터 구조를 쓰든 중요하지 않다.

어차피 배열을 내부 구조에서 사용하는데 왜 굳이 스택을 써야 하는 건가?
1. 제약을 같고 있기에 잠재적 버그를 막을 수 있다.
2. 스택과 같은 데이터 구조는 문제 해결하는 새로운 사고 모델을 제공한다.

스택은 마지막에 들어 온 데이터부터 먼저 처리해야 할 때 이상적이다.

스택 응용(계산기)
```java
import java.util.ArrayDeque;
import java.util.Deque;

public class StackCalculator{
	public static void main(String[] args){
		String expression = "3 + 5 * (2 - 1)";
		
		int result = calculate(expression);
		
		System.out.println(result);
	}
	
	static int calculate(String expression){
		Deque<Integer> numbers = new ArrayDeque<>();
		Deque<Character> operators = new ArrayDeque<>();
		
		for(int i = 0; i <expression.charAt(i); i++){
			char c = expression.charAt(i);
			
			// 공백 무시
			if(c==' '){
				return;
			}
			
			// 숫자
			if(Character.isDigit(c)){
				int num = 0;
				
				// 여러 자리 숫자 처리
				while(i < expression.length() && Character.isDigit(expression.charAt(i) - '0')){
					num = num*10 + expression.charAt(i) -'0';
					i++;
				}
				i--;
				numbers.push(num);
			}
			// 여는 괄호
			else if(c=='('){
				operators.push(c);
			}
			// 닫는 괄호
			else if(c==')'){
				while(operators.peek() != '('){
					calculateTop(numbers,operators);
				}
				operators.pop(); // '('제거
			}
			// 연산자
			else if(isOperator(c)){
				while(!operators.isEmpty() && operators.peek() != '(' && priority(operators.peek()) >= priority(c)){
					calculateTop(numbers, operators);
				}
				operators.push(c);
			}
		}
		// 남은 연산 수행
		while(!operators.isEmpty()){
			calculateTop(numbers,operators);
		}
		return numbers.pop();
	}
	static void calculateTop(Deque<Integer> numbers, Deque<Character> operators){
		int b = numbers.pop();
		int a = numbers.pop();
		
		char operator = operators.pop();
		
		int result;
		
		switch(operator){
			case '+':
				result = a + b;
				break;
			case '-':
				result = a - b;
				break;
			case '*':
				result = a * b;
				break;
			case '/':
				result = a / b;
				break;
			default:
				throw new IllegalArgumentException();
		}
		numbers.push(result);
	}
	static boolean isOperator(char c){
		return c == '+' || c == '-' || c == '*' || c == '/';
	}
	
	static int priority(char operator){
		if(operator == '+' || operator == '-'){
			return 1;
		}
		if(operator == '*' || operator == '/'){
			return 2;
		}
		return 0;
	}
}
```
