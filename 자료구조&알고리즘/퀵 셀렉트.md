무작위로 정렬된 [[배열]]이 있을 때, n번째로 작은(또는 큰) 값을 찾아보는 방법으로는 정렬을 한 뒤 찾는 방법이 있지만 이 알고리즘은 [[퀵 정렬]]을 사용하더라도 `nO(log n)`이 걸린다. 
퀵 셀렉트는 더 좋은 방법으로 원하는 인덱스를 찾을 수 있다.

퀵 셀렉트도 역시 분할에 기반하며, [[퀵 정렬]]과 [[이진 검색]]의 하이브리드 정도로 생각할 수 있다.
퀵 셀렉트는 전체 배열을 정렬하지 않고도 올바른 값을 찾을 수 있다.

1. 퀵 정렬 처럼 피벗을 선택한다.
2. 피벗을 기준으로 분할한다.
3. 찾고 싶은 값과 피벗을 비교하여 값이 포함된 하위 배열만 집중한다. (피벗 왼쪽 or 피벗 or 피벗 오른쪽)
4. 이를 값을 찾을 때 까지 반복

퀵 정렬과의 차이는 퀵 정렬은 왼쪽 하위배열과 오른쪽 하위배열 모두 정렬을 하지만 퀵 셀렉트는 필요 없는 배열을 버린다.

퀵 셀렉트의 평균 시간복잡도는 `O(n)`이고, 최악의 경우 O($n^2$)이다.

```java
public class QuickSelect{

	// k번째로 작은 값을 찾는다.
	// k는 1부터 시작
	public static int quickSelect(int[] arr, int left, int right, int k){
	
		if(k < 1 || k > arr.length){
			throw new IllegalArgumentException("k의 범위가 잘못되었습니다.");
		}
		
		if (left <= right){
			// Two-pointer + pivot swap 방식으로 분할
			int pivotIndex = partition(arr,left, right);
			// k번째 원소의 실제 인덱스는 k - 1
			int targetIndex = k - 1;
			// 피벗이 찾고 있는 위치에 도달
			if(pivotIndex == targetIndex){
				return arr[pivotIndex];
			}
			// 찾는 위치가 피벗의 왼쪽
			else if (targetIndex < pivotIndex){
				return quickSelect(arr, left, pivotIndex - 1, k);
			}
			// 찾는 위치가 피벗의 오른쪽
			else {
				return quickSelect(arr, pivotIndex + 1, right, k);
			}
		
		}
	
	}
	
	// Two-pointer + 마지막 pivot swap 분할
	private static int partition(int[] arr, int left, int right){
		// 가장 오른쪽 원소를 피벗으로 선택
		int pivot = arr[right];
		
		int leftPointer = left;
		int rightPointer = right;
		
		while(true){
			// 왼쪽 포인터
			// 피벗보다 크거나 같은 값을 만날 때까지 이동
			while(leftPointer <= rightPointer && arr[leftPointer] < pivot){
			leftPointer++;
			}
			// 오른쪽 포인터
			// 피벗보다 작거나 같은 값을 만날 때 까지 이동
			while(leftPointer <= rightPointer && arr[rightPointer] > pivot){
			rightPointer--;
			}
			
			// 두 포인터가 만나거나 교차했다면 종료
			if(leftPointer >= rightPointer){
				break;
			}
			// 잘못된 위치의 두 원소 교환
			swap(arr, leftPointer, rightPointer);
			// 중복값이 있을 때 무한 반복 방지
			leftPointer++;
			rightPointer--;
		}
		// 왼쪽 포인터 위치와 피벗을 교환
		swap(arr, leftPointer, right);
		// 피벗의 최종 위치 변환
		return leftPointer;
	}
	
	private static void swap(int[] arr, int i, int j){
		int temp = arr[i];
		arr[i] = arr[j];
		arr[j] = temp;
	}
	
	public static void main(String[] args){
		int[] arr = {7,2,9,4,1,5};
		int k = 3;
		
		int result = quickSelect(arr, 0, arr.length -1, k);
		System.out.println(k + "번째로 작은 값: " + result);
	}

}
```
