분할 정복(Divide and Conquer)은 큰 문제를 **작은 문제로 쪼개서 각각 해결한 뒤, 그 결과를 합쳐 원래 문제를 해결하는 알고리즘 설계 방식**

분할 정복은 재귀와 자주 같이 사용됩니다.

대표적인 문제 
[쿼드압축 후 개수 세기](https://school.programmers.co.kr/learn/courses/30/lessons/68936?utm_source=chatgpt.com)

```java
class Solution{
	int[] answer = new int[2];
	public int[] solution(int[][] arr){
		divide(arr,0,0,arr.length);
		return answer;
	}
	
	void divide(int[][] arr, int row, int col, int size){
		// 현재 영역이 모두 같은 숫자인지 확인
		if(isSame(arr,row,col,size)){
			answer[arr[row][col]++];
			return;
		}
		
		// 같은 숫자가 아니라면 4등분
		int half = size / 2;
		
		divide(arr,row,col,half); // 왼쪽 위
		divide(arr,row,col + half,half); // 오른쪽 위
		divide(arr,row + half, col, half); // 왼쪽 아래
		divide(arr,row + half, col + half, half); // 오른쪽 아래
	}
	boolean isSame(int[][] arr, int row, int col, int size){
		int value = arr[row][col];
		
		for(int i = row; i < row + size; i++){
			for(int j = col; j < col + size; j++){
				if(arr[i][j] != value){
					return false;
				}
			}
		}
		return true;
	}
}
```
